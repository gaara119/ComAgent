执行的 SQL

WITH yearly AS (SELECT CAST(DATE_TRUNC('YEAR', "invoice_date") AS DATE) AS yr, SUM("total") AS revenue FROM "public"."invoice" WHERE "billing_country" = 'USA' AND "invoice_date" >= '2023-01-01' AND "invoice_date" < '2025-01-01' GROUP BY 1) SELECT (SUM(CASE WHEN yr = '2024-01-01' THEN revenue ELSE 0 END) - SUM(CASE WHEN yr = '2023-01-01' THEN revenue ELSE 0 END)) / NULLIF(SUM(CASE WHEN yr = '2023-01-01' THEN revenue ELSE 0 END), 0) AS yoy_growth_rate FROM yearly
