执行的 SQL

WITH revenue_by_year AS (SELECT CASE WHEN "invoice_date" >= CAST('2023-01-01 00:00:00' AS TIMESTAMP) AND "invoice_date" < CAST('2024-01-01 00:00:00' AS TIMESTAMP) THEN 'base' WHEN "invoice_date" >= CAST('2024-01-01 00:00:00' AS TIMESTAMP) AND "invoice_date" < CAST('2025-01-01 00:00:00' AS TIMESTAMP) THEN 'current' END AS period, SUM("total") AS revenue FROM "public"."invoice" WHERE "billing_country" = 'USA' AND "invoice_date" >= CAST('2023-01-01 00:00:00' AS TIMESTAMP) AND "invoice_date" < CAST('2025-01-01 00:00:00' AS TIMESTAMP) GROUP BY period) SELECT (SUM(CASE WHEN period = 'current' THEN revenue END) - SUM(CASE WHEN period = 'base' THEN revenue END)) / NULLIF(SUM(CASE WHEN period = 'base' THEN revenue END), 0) * 100 AS yoy_growth_rate FROM revenue_by_year
